### Guarde o valor do endpoint
export API_ENDPOINT=$(aws cloudformation describe-stacks --stack-name ws-serverless-patterns-users --output text --query 'Stacks[0].Outputs[?OutputKey==`APIEndpoint`].OutputValue')
echo "API endpoint: $API_ENDPOINT"

### Teste a API

### Crie uma variável do ambiente do AUTH_BASE64.

export AUTH_BASE64=$(<base64_encoded_username)

### Teste a API com o ID_BASE64

curl -H "Authorization: Basic <base64_encoded_username:password>" $API_ENDPOINT/users
ou
curl $API_ENDPOINT/users -H "Authorization: Basic $ID_BASE64"

### Adicionar dados via API

         --header "Authorization: Basic $ID_BASE64" --header "Content-Type: application/json"
